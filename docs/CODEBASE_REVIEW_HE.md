~~~python
from src.course_parser import parse_group
from src.solver import groups_conflict

def group(identifier, meetings):
    return parse_group({
        "group_id": identifier, "lecturer": "L", "priority": "medium",
        "sessions": [
            {"day": day, "start": start, "end": end, "mandatory": True}
            for day, start, end in meetings
        ],
    })

a = group("a", [(1, "08:00", "09:00"), (2, "08:00", "09:00")])
b = group("b", [(2, "08:30", "09:30")])
assert groups_conflict(a, b)  # נכשל בקוד שנבדק ב־12.10.2026
~~~
